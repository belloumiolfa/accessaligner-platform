import { Component } from "@angular/core";
import { NgxSpinnerService } from "ngx-spinner";
import { UserService } from "../../../Core/Services/UserService/user.service";
import { GridUsersComponent } from "../grid-users/grid-users.component";
import { HandleErrorsService } from "../../../Core/Helpers/handle-errors.service";

@Component({
  selector: "app-archive-doctors",
  standalone: true,
  imports: [GridUsersComponent],
  templateUrl: "./archive-doctors.component.html",
  styleUrl: "./archive-doctors.component.css",
})
export class ArchiveDoctorsComponent {
  doctors$: any[] = [];
  photoMap: any[] = [];
  errors: any;

  constructor(
    private userServices: UserService,
    private spinner: NgxSpinnerService,
    private handleErrors: HandleErrorsService
  ) {
    this.spinner.show();
    this.userServices.getDoctors().subscribe(
      (data) => {
        this.doctors$ = data.filter(
          (doctor: { userStatus: { status: { name: string } } }) =>
            doctor.userStatus.status.name === "BLOCKED"
        );

        this.spinner.hide();
      },
      (err) => {
        this.errors = this.handleErrors.handleError(err);
        this.spinner.hide();
      }
    );
  }
}
