import { CommonModule } from '@angular/common';
import { Component } from '@angular/core';
import { RouterModule, RouterOutlet } from '@angular/router';
import { NgbCollapseModule, NgbDropdown } from '@ng-bootstrap/ng-bootstrap';
import { AccordionModule } from 'ngx-bootstrap/accordion';
import { MiniLeftbarComponent } from '../mini-leftbar/mini-leftbar.component';
import { LeftSidebarComponent } from '../left-sidebar/left-sidebar.component';
import { loggedInUser } from '../../Core/Helpers/utils';
import { UtilsService } from '../../Auth/Helpers/utils.service';
import { NgxSpinnerService } from 'ngx-spinner';
import { HandleErrorsService } from '../../Core/Helpers/handle-errors.service';
import { AuthService } from '../../Core/Services/AuthService/auth.service';
import { User } from '../../Core/Models/user.models';
import { AppService } from '../../Core/Services/app.service';
import { SafeUrl } from '@angular/platform-browser';
import { TreatmentService } from '../../Core/Services/TreatService/treatment.service';
import { PatientService } from '../../Core/Services/PatientService/patient.service';

export interface MenuItem {
  id?: number;
  key?: string;
  label?: string;
  icon?: string;
  url?: string;
  collapsed?: boolean;
  children?: any;
  isTitle?: boolean;
  badge?: any;
  parentKey?: number;
  class?: string;
}

@Component({
  selector: 'app-layout-container',
  standalone: true,
  imports: [
    RouterOutlet,
    CommonModule,
    AccordionModule,
    MiniLeftbarComponent,
    NgbCollapseModule,
    NgbDropdown,
    RouterModule,
    LeftSidebarComponent,
  ],
  templateUrl: './layout-container.component.html',
  styleUrl: './layout-container.component.css',
})
export class LayoutContainerComponent {
  showMobileMenu: boolean = true;

  menuItems: MenuItem[] = [];
  activeMenuItems: string[] = [];
  chunkSize: number = 7;
  div: HTMLElement | null | undefined;
  isCollapsed: any = true;

  user$!: User;
  profilePhoto$!: any;

  errors!: any;
  imageUrl!: SafeUrl;

  constructor(
    private utils: UtilsService,
    private authService: AuthService,
    private handleErrors: HandleErrorsService,
    private spinner: NgxSpinnerService,
    private appService: AppService,
    private TreatmentService: TreatmentService,
    private PatientService: PatientService,
    private utilsService: UtilsService
  ) {
    this.appService.getUser$.subscribe((data) => (this.user$ = data));
  }

  ngOnInit(): void {
    this.spinner.show();
    this.authService
      .getUserById(this.utils.getDecodedAccessToken(loggedInUser()).userId)
      .subscribe(
        (data) => {
          this.appService.setUser$(data);

          this.setLengthPatients(data);
          this.setLengthTreatments(data);
          this.spinner.hide();
        },
        (err) => {
          this.errors = this.handleErrors.handleError(err);
        }
      );
  }

  setLengthTreatments(data: any) {
    this.TreatmentService.getTreatments().subscribe((dataTreatments) => {
      let length = 0;
      let lengthArchive = 0;
      if (this.utilsService.isDoctor(data)) {
        length = dataTreatments.filter(
          (treatment: { patient: any; status: string; doctor: { id: any } }) =>
            treatment.patient?.doctor?.id === data.id &&
            treatment.status !== 'FINISHED' &&
            treatment.status !== 'CANCELED' &&
            treatment.status !== 'UNQUALIFIED'
        ).length;

        lengthArchive = dataTreatments.filter(
          (treatment: { patient: any; status: string; doctor: { id: any } }) =>
            ((treatment.patient?.doctor?.id === data.id) === data.id &&
              treatment.status === 'FINISHED') ||
            treatment.status === 'CANCELED' ||
            treatment.status === 'UNQUALIFIED'
        ).length;
      } else {
        length = dataTreatments.filter(
          (treatment: { status: string; doctor: { id: any } }) =>
            treatment.status !== 'FINISHED' &&
            treatment.status !== 'CANCELED' &&
            treatment.status !== 'UNQUALIFIED'
        ).length;

        lengthArchive = dataTreatments.filter(
          (treatment: { status: string; doctor: { id: any } }) =>
            treatment.status === 'FINISHED' ||
            treatment.status === 'CANCELED' ||
            treatment.status === 'UNQUALIFIED'
        ).length;
      }

      this.appService.setNbrTreatments(length);
      this.appService.setNbrTreatmentsArchive(lengthArchive);
    });
  }

  setLengthPatients(data: any) {
    this.PatientService.getPatients().subscribe((dataPatients) => {
      let length = 0;
      if (this.utilsService.isDoctor(data)) {
        length = dataPatients.filter(
          (patient: { doctor: { id: any } }) => patient.doctor.id === data.id
        ).length;
      } else {
        length = dataPatients.length;
      }
      this.appService.setNbrPatients(length);
    });
  }
}
