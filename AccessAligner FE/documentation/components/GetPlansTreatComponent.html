import { CommonModule } from '@angular/common';
import { Component } from '@angular/core';
import { FormsModule, ReactiveFormsModule } from '@angular/forms';
import { RouterModule } from '@angular/router';
import { ModalService } from '../../../Core/Helpers/modal.service';
import { TeamInfoComponent } from '../../../Shared/Elements/team-info/team-info.component';
import { AppService } from '../../../Core/Services/app.service';
import { PlanService } from '../../../Core/Services/PlanService/plan.service';
import Swal from 'sweetalert2';
import { NgxSpinnerService } from 'ngx-spinner';
import { HandleAlertsService } from '../../../Core/Helpers/handle-alerts.service';
import { StatusClassService } from '../../../Core/Helpers/status-class.service';
import { getDate } from '../../../Core/Helpers/utils';
import { HandleErrorsService } from '../../../Core/Helpers/handle-errors.service';
import { UtilsService } from '../../../Auth/Helpers/utils.service';
@Component({
  selector: 'app-get-plans-treat',
  standalone: true,
  imports: [
    CommonModule,
    RouterModule,
    ReactiveFormsModule,
    FormsModule,
    TeamInfoComponent,
  ],
  templateUrl: './get-plans-treat.component.html',
  styleUrl: './get-plans-treat.component.css',
})
export class GetPlansTreatComponent {
  treatment$!: any;

  show: any;
  plan$!: any;
  errors: any;
  user$!: any;
  spinnerShow: boolean = true;
  constructor(
    private modalService: ModalService,
    private appService: AppService,
    private planService: PlanService,
    private spinner: NgxSpinnerService,
    private handleAlerts: HandleAlertsService,
    private statusService: StatusClassService,
    private handleErrors: HandleErrorsService,
    public utils: UtilsService
  ) {
    this.appService.getUser$.subscribe((data) => (this.user$ = data));

    this.appService.getPlan$.subscribe((data) => {
      this.plan$ = data;
      this.spinnerShow = false;
    });

    this.appService.getTreatment$.subscribe((data) => {
      this.treatment$ = data;
    });
  }
  acceptedPlan() {
    console.log(
      this.treatment$.plans.filter(
        (e: any) => e.status !== 'NEW' && e.status !== 'REJECTED'
      )[0].status
    );

    return this.treatment$.plans.filter(
      (e: any) => e.status !== 'NEW' && e.status !== 'REJECTED'
    )[0];
  }

  getDate(date: any) {
    return getDate(date);
  }

  onOpenModalPlan(plan: any) {
    this.appService.setPlan$(plan);
    this.modalService.openDisplayPlanModel(plan, this.treatment$);
  }

  getSatusClass(status: string): string {
    return this.statusService.getClassStatus(status);
  }

  getPCClass(status: string): string {
    return this.statusService.getClassPC(status);
  }

  getPCByStatus(status: string) {
    return this.statusService.getPCByStatus(status);
  }

  onDelete(planId: any) {
    this.planService.deletePlan(planId).subscribe(
      (data) => {
        this.appService.setTreatment({
          ...this.treatment$,
          plans: [...this.treatment$.plans.filter((p: any) => p.id !== planId)],
        });

        this.spinner.hide();
        this.handleAlerts.handleSweetAlert(
          'Treatment has been successfully deleted. ',
          'success',
          false
        );
      },
      (err) => {
        console.log(err);

        this.spinner.hide();
        this.errors = this.handleErrors.handleError(err);

        this.handleAlerts.handleSweetAlert(
          'Check your data input carefully.',
          'error',
          false
        );
      }
    );
  }

  deleteConfrim(id: any, status: string) {
    if (status === 'DELIVERED' || status === 'PRODUCTION') {
      Swal.fire({
        title: 'Plan in ' + status + ' status',
        text: "It can't be deleted",
        icon: 'warning',

        confirmButtonColor: '#DD6B55',
        cancelButtonColor: '#b9b9b9',
        confirmButtonText: 'Ok',
        customClass: {
          popup: 'sweet-alert', // corrected class name
        },
      });
    } else {
      Swal.fire({
        title: 'Are you sure?',
        text: "You won't be able to revert this!",
        icon: 'warning',
        showCancelButton: true,
        confirmButtonColor: '#DD6B55',
        cancelButtonColor: '#b9b9b9',
        confirmButtonText: 'Yes, delete it!',
        customClass: {
          popup: 'sweet-alert',
        },
      }).then((result) => {
        if (result.isConfirmed) {
          this.spinner.show();

          this.onDelete(id);
        }
      });
    }
  }

  onUpdateSatusConfirm(planId: any, status: string) {
    Swal.fire({
      title: 'Are you sure to ' + status + ' plan ?',
      text: '',
      icon: 'warning',
      showCancelButton: true,
      confirmButtonColor: '#DD6B55',
      cancelButtonColor: '#b9b9b9',
      confirmButtonText: 'Yes, update it!',
      customClass: {
        popup: 'sweet-alert',
      },
    }).then((result) => {
      if (result.isConfirmed) {
        this.spinner.show();

        this.onUpdateStatus(planId, status);
      }
    });
  }

  onUpdateStatus(planId: any, status: string) {
    this.planService.updateStatus(planId, status).subscribe(
      (data) => {
        this.appService.setTreatment({
          ...this.treatment$,
          plans: [
            ...this.treatment$.plans.filter((p: any) => p.id !== data.id),
            data,
          ],
        });
        this.handleAlerts.handleSweetAlert(
          'Plan has been updated successfully.',
          'success',
          false
        );
        this.spinner.hide();
      },
      (err) => {
        console.log(err);

        this.spinner.hide();
        this.errors = this.handleErrors.handleError(err);

        this.handleAlerts.handleSweetAlert(
          'Check your data input carefully.',
          'error',
          false
        );
      }
    );
  }
}
