import { CommonModule } from "@angular/common";
import { AfterViewInit, Component, Input, OnInit } from "@angular/core";
import { AppService } from "../../../Core/Services/app.service";
import { NgbActiveModal } from "@ng-bootstrap/ng-bootstrap";
import { TeamInfoComponent } from "../../../Shared/Elements/team-info/team-info.component";
import { ModalService } from "../../../Core/Helpers/modal.service";
import { NgxSpinnerService } from "ngx-spinner";
import { NgxSpinnerModule } from "ngx-spinner";
import { PlanService } from "../../../Core/Services/PlanService/plan.service";
import { FormBuilder, FormControl, ReactiveFormsModule } from "@angular/forms";
import { getDate } from "../../../Core/Helpers/utils";
import { StatusClassService } from "../../../Core/Helpers/status-class.service";
import { HandleAlertsService } from "../../../Core/Helpers/handle-alerts.service";
import { HandleErrorsService } from "../../../Core/Helpers/handle-errors.service";
import { UtilsService } from "../../../Auth/Helpers/utils.service";
import Swal from "sweetalert2";

declare var $: any;
@Component({
  selector: "app-modal-plan-details",
  standalone: true,
  imports: [
    CommonModule,
    TeamInfoComponent,
    NgxSpinnerModule,
    ReactiveFormsModule,
  ],
  templateUrl: "./modal-plan-details.component.html",
  styleUrl: "./modal-plan-details.component.css",
})
export class ModalPlanDetailsComponent implements OnInit, AfterViewInit {
  @Input() plan: any;
  @Input() treatment: any;
  plan$!: any;

  existedFilesPhotos: any[] = [];
  existedFilesVedios: any[] = [];
  existedFilesReports: any[] = [];
  productionDelay: any;
  treatments$!: any[];

  myGroup: any;
  deliveryDate!: any;
  productionForm!: any;
  feedBackForm!: any;

  showPhotosSpinner: boolean = false;
  showVideosSpinner: boolean = false;
  showReprotsSpinner: boolean = false;
  errors: any;
  user$!: any;

  // data : any = { existedPhotos: [] , showSpinner:false} ;
  constructor(
    private activeModal: NgbActiveModal,
    private modalService: ModalService,
    private planService: PlanService,
    private spinner: NgxSpinnerService,
    private appService: AppService,
    private statucClassService: StatusClassService,
    private formBuilder: FormBuilder,
    private handleAlerts: HandleAlertsService,
    private handleErrors: HandleErrorsService,
    public utils: UtilsService
  ) {
    this.appService.getUser$.subscribe((data) => (this.user$ = data));

    this.deliveryDate = new FormControl("");
    this.productionForm = new FormControl("");
    this.feedBackForm = new FormControl("");
  }

  close() {
    this.activeModal.close();
  }

  ngOnInit() {
    if (this.plan) {
      if (this.plan.photos.length !== 0) {
        this.existedFilesPhotos = this.appService.getFilesPlan(
          this.plan.photos,
          this.treatment.id
        );
      }

      if (this.plan.videos.length !== 0) {
        this.existedFilesVedios = this.appService.getFilesPlan(
          this.plan.videos,
          this.treatment.id
        );
      }

      if (this.plan.reports.length !== 0) {
        this.getRepports(this.plan.reports, this.treatment.id, this.plan.id);
      }
    }
  }

  getRepports(repports: any[], treatId: Number, planId: Number) {
    let repportsWithURL: any[] = [];

    repports?.forEach((element: any) => {
      this.planService.getRepports(element.id, treatId, planId).subscribe(
        (data) => {
          element = { ...element, blob: data };
          repportsWithURL.push(element);
        },
        (err) => {
          console.log(err);

          this.spinner.hide();
          this.errors = this.handleErrors.handleError(err);

          this.handleAlerts.handleSweetAlert(
            "Check your data input carefully.",
            "error",
            false
          );
        }
      );
    });

    this.existedFilesReports = repportsWithURL;
  }

  ngAfterViewInit(): void {
    $("#datetimepicker")
      .bootstrapMaterialDatePicker({
        weekStart: 0,
        time: false,
        minDate: new Date(),
      })
      .on("change", (e: any, date: { format: (arg0: string) => any }) => {
        const formattedDate = date.format("YYYY-MM-DD");
        this.deliveryDate?.setValue(formattedDate);
      });
  }

  onOpenPdf(file: any) {
    window.open(URL.createObjectURL(file.blob));
  }

  updateTreatmentPlan(data: any) {
    this.appService.setPlan$(data);
    this.appService.setTreatment({
      ...this.treatment,
      plans: [
        ...this.treatment.plans.filter((p: any) => p.id !== data.id),
        data,
      ],
    });

    this.spinner.hide();
    //this.close();
  }

  displayItem(item: any) {
    this.modalService.openDisplayPhtotoPlanModel(item);
  }

  onUpdateStatus(planId: Number, status: string) {
    this.planService.updateStatus(planId, status).subscribe(
      (data) => {
        this.updateTreatmentPlan(data);
        // close spinner

        // update feedback service
        if (!!this.feedBackForm.value) {
          this.spinner.show();

          this.planService
            .updateFeedBack(planId, this.feedBackForm.value)
            .subscribe(
              (data) => {
                this.spinner.hide();

                this.updateTreatmentPlan(data);
                this.handleAlerts.handleSweetAlert(
                  "Your feedback has been added successfully.",
                  "success",
                  false
                );
              },
              (err) => {
                console.log(err);

                this.spinner.hide();
                this.errors = this.handleErrors.handleError(err);

                this.handleAlerts.handleSweetAlert(
                  "Check your data input carefully.",
                  "error",
                  false
                );
              }
            );
        }
        this.spinner.hide();
        this.handleAlerts.handleSweetAlert(
          "Plan status has been updated successfully.",
          "success",
          false
        );
      },
      (err) => {
        console.log(err);

        this.spinner.hide();
        this.errors = this.handleErrors.handleError(err);

        this.handleAlerts.handleSweetAlert(
          "Check your data input carefully.",
          "error",
          false
        );
      }
    );
  }

  // deleveryForm: any = new FormControl('');
  onUpdateDeleveryDate(planId: Number, date: Date, status: string) {
    Swal.fire({
      title: "Are you sure to " + status + " plan ?",
      text: "",
      icon: "warning",
      showCancelButton: true,
      confirmButtonColor: "#DD6B55",
      cancelButtonColor: "#b9b9b9",
      confirmButtonText: "Yes, update it!",
      customClass: {
        popup: "sweet-alert",
      },
    }).then((result) => {
      if (result.isConfirmed) {
        this.spinner.show();

        this.planService.updateDateDelevery(planId, new Date(date)).subscribe(
          (data) => {
            this.onUpdateStatus(planId, status);
          },
          (err) => {
            console.log(err);

            this.spinner.hide();
            this.errors = this.handleErrors.handleError(err);

            this.handleAlerts.handleSweetAlert(
              "Check your data input carefully.",
              "error",
              false
            );
          }
        );
      }
    });
  }

  onStartProduction(planId: Number) {
    Swal.fire({
      title: "Are you sure to updaet the plan ?",
      text: "",
      icon: "warning",
      showCancelButton: true,
      confirmButtonColor: "#DD6B55",
      cancelButtonColor: "#b9b9b9",
      confirmButtonText: "Yes, update it!",
      customClass: {
        popup: "sweet-alert",
      },
    }).then((result) => {
      if (result.isConfirmed) {
        this.spinner.show();

        this.planService
          .updateProductionDelay(planId, this.productionForm.value)
          .subscribe(
            (data) => {
              this.onUpdateStatus(planId, "PRODUCTION");
            },
            (err) => {
              console.log(err);

              this.spinner.hide();
              this.errors = this.handleErrors.handleError(err);

              this.handleAlerts.handleSweetAlert(
                "Check your data input carefully.",
                "error",
                false
              );
            }
          );
      }
    });
  }

  convertDate(date: Date) {
    return getDate(date);
  }

  getStatusClass(status: any): any {
    return this.statucClassService.getClassStatus(status);
  }
}
