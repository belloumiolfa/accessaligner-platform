import { Component, ElementRef, Input, ViewChild } from "@angular/core";
import { EstimateService } from "../../../Core/Services/EstimateService/estimate.service";
import html2canvas from "html2canvas";
import jsPDF from "jspdf";
import { AppService } from "../../../Core/Services/app.service";
import { NgbActiveModal } from "@ng-bootstrap/ng-bootstrap";
import { getDate } from "../../../Core/Helpers/utils";
import { HandleAlertsService } from "../../../Core/Helpers/handle-alerts.service";
import { HandleErrorsService } from "../../../Core/Helpers/handle-errors.service";
import { NgxSpinnerService, Spinner } from "ngx-spinner";

@Component({
  selector: "app-content-devis-treat",
  standalone: true,
  imports: [],
  templateUrl: "./content-devis-treat.component.html",
  styleUrl: "./content-devis-treat.component.css",
})
export class ContentDevisTreatComponent {
  @Input() treatment: any;
  @Input() dataDevis: any;
  errors!: any;
  email: string = "contact@accessaligner.com";

  constructor(
    private estimateService: EstimateService,
    private appService: AppService,
    private activeModal: NgbActiveModal,
    private handleAlerts: HandleAlertsService,
    private handleErrors: HandleErrorsService,
    private spinner: NgxSpinnerService
  ) {}
  getCurrentDate() {
    console.log("dataa currenttttttt ", new Date());
    return getDate(new Date());
  }

  @ViewChild("contentToExport", { static: false }) contentToExport!: ElementRef;

  //exportAndUploadPdf
  exportToPdf() {
    const content = this.contentToExport.nativeElement;
    let role = this.treatment.plans.length > 0 ? "final_estim" : "init_estim";
    this.generatePdf(content, role)
      .then((data: any) => {
        const fileName = role + "_treat_" + this.treatment.id + ".pdf";

        //Upload pdf service
        this.estimateService
          .uploadPdfFile(data, this.treatment.id, role, fileName)
          .subscribe(
            (data) => {
              this.appService.setTreatment(data);
              this.handleAlerts.handleSweetAlert(
                "File has been uploaded successfully.",
                "success",
                false
              );
            },
            (err) => {
              console.log(err);

              this.spinner.hide();
              this.errors = this.handleErrors.handleError(err);

              this.handleAlerts.handleSweetAlert(
                "Check your data input carefully.",
                "error",
                false
              );
            }
          );
      })
      .catch((error) => {
        console.error("Error generating PDF:", error);
        this.handleAlerts.handleSweetAlert(
          "Error generating PDF.",
          "error",
          false
        );
      });

    this.activeModal.close();
  }
  generatePdf(htmlContent: HTMLElement, fileName: any): Promise<any> {
    return new Promise((resolve, reject) => {
      html2canvas(htmlContent)
        .then((canvas) => {
          // Initialize jsPDF
          const pdf = new jsPDF("p", "mm", "a4");
          const imgData = canvas.toDataURL("image/png");
          pdf.addImage(imgData, "PNG", 0, 0, 210, 297); // landscape A4 dimensions

          // Save pdf
          pdf.save(fileName + ".pdf");

          // Generate Blob
          var blobPDF = new Blob([pdf.output("blob")], {
            type: "application/pdf",
          });
          resolve(blobPDF);
        })
        .catch((err) => {
          reject(err);
        });
    });
  }
}
